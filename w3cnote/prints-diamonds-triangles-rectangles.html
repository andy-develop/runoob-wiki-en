here, if you remove the if-else condition check, it will be solid." for i in range(rows):# Variable i controls the number of lines for j in range(rows - i):#(1,rows-i) print " ", j += 1 for k in range(2 * i - 1):#(1,2*i) if k == 0 or k == 2 * i - 2: print "*", else: print " ", k += 1 print "\n" i += 1 # The lower half of the diamond for i in range(rows): for j in range(i):#(1,rows-i) print " ", j += 1 for k in range(2 * (rows - i) - 1):#(1,2*i) if k == 0 or k == 2 * (rows - i) - 2: print "*", else: print " ", k += 1 print "\n" i += 1 #Solid square print "Solid square" for i in range(0, rows): for k in range(0, rows): print " * ", # Note the "," here, it must not be omitted, it can prevent line breaks k += 1 i += 1 print "\n" #Hollow square print "Hollow square" for i in range(0, rows): for k in range(0, rows): if i != 0 and i != rows - 1: if k == 0 or k == rows - 1: # Because visually it looks more like a square, spaces are added on both sides here to increase the distance. print " * ", # Note the "," here, it must not be omitted, it can prevent line breaks else: print " ", # There are three spaces here else: print " * ", # Here spaces are added on both sides of * k += 1 i += 1 print "\n"
