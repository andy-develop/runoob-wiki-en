C Standard Library -<stdlib.h>

简介

stdlib .hThe header file defines four variable types, some macros, and various general utility functions.

<stdlib.h>is a header file in the C standard library that provides many general-purpose utility functions, including memory allocation, process control, sorting and searching, and string conversion, among others.

实例

以下是使用<stdlib.h>Examples of some functions in.

Dynamic memory allocation:

实例

#include <stdio.h>
#include <stdlib.h>

int main() {
    int *array = (int*)malloc(10 * sizeof(int));
    if (array == NULL) {
        perror("Memory allocation failed");
        return 1;
    }

    for (int i = 0; i < 10; i++) {
        array[i] = i;
        printf("%d ", array[i]);
    }
    printf("\n");

    free(array);
    return 0;
}

Pseudo-random number generation:

实例

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main() {
    srand(time(NULL));
    for (int i = 0; i < 10; i++) {
        printf("%d ", rand() % 100);
    }
    printf("\n");

    return 0;
}

String conversion:

实例

#include <stdio.h>
#include <stdlib.h>

int main() {
    const char *str = "12345";
    int num = atoi(str);
    printf("The integer value is %d\n", num);

    return 0;
}

快速排序:

实例

#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int main() {
    int array[] = {5, 2, 9, 1, 5, 6};
    int size = sizeof(array) / sizeof(array[0]);

    qsort(array, size, sizeof(int), compare);

    for (int i = 0; i < size; i++) {
        printf("%d ", array[i]);
    }
    printf("\n");

    return 0;
}

注意事项

  • 使用malloc、calloc 和 reallocThe allocated memory must be freed usingfreeto avoid memory leaks.
  • atoi、atol 和 atofThe function does not provide error checking; ensure the input format is correct when using it. It is recommended to usestrtol、strtoul 和 strtodto perform safer conversions.
  • qsort 和 bsearchThe function requires a custom comparison function to handle different data types and sorting requirements.
