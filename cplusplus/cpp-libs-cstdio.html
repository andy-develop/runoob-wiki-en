C++ 标准库 <cstdio>

<cstdio>It is a header file in the C++ standard library. It contains the C++ wrapper of the C standard I/O library and is mainly used for file input and output operations.

<cstdio>The library defines a set of functions for performing input and output operations. These functions can be used to read and write files and the console.

语法

在使用<cstdio>Before using the functions in the library, you need to include this header file at the top of the C++ program:

#include <cstdio>

常用函数

<cstdio>The library contains many functions for file I/O. The following are some commonly used functions:

  • fopen: Open file.
  • fclose: Close file.
  • fread: Read data from a file.
  • fwrite: Write data to a file.
  • fprintf: Write formatted output to a file.
  • fscanf: Read formatted input from a file.
  • fgetc: Read a character from a file.
  • fputc: Write a character to a file.
  • fgets: Read a line from a file.
  • fputs: Write a line to a file.

实例

1. Open and close files:

实例

#include <cstdio>

int main() {
    FILE *file = fopen("example.txt", "w"); // 打开文件用于写入
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }
    fclose(file); // 关闭文件
    return 0;
}

2. Write data to a file:

实例

#include <cstdio>

int main() {
    FILE *file = fopen("example.txt", "w");
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }
    fprintf(file, "Hello, World!n");
    fclose(file);
    return 0;
}

在 "example.txt" 文件中写入了 "Hello, World!"。

3. Read data from a file:

实例

#include <cstdio>
#include <iostream>

int main() {
    FILE *file = fopen("example.txt", "r");
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }
    char buffer[100];
    while (fgets(buffer, 100, file) != NULL) {
        std::cout << buffer;
    }
    fclose(file);
    return 0;
}

Read from the "example.txt" file and output "Hello, World!".

4. Use fscanf and fprintf for formatted input and output:

实例

#include <cstdio>

int main() {
    FILE *file = fopen("data.txt", "w");
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }
    fprintf(file, "%d %fn", 42, 3.14159);
    fclose(file);

    file = fopen("data.txt", "r");
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }
    int number;
    float pi;
    fscanf(file, "%d %f", &number, &pi);
    fclose(file);

    std::printf("Number: %d, Pi: %fn", number, pi);
    return 0;
}

输出结果：Number: 42, Pi: 3.141590

fopen 和 fclose

Used to open and close files.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        // 文件操作
        fclose(file);
    } else {
        // 处理错误
    }
    return 0;
}

fread 和 fwrite

Used to read and write data from and to files.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.bin", "wb");
    if (file) {
        int data = 12345;
        fwrite(&data, sizeof(data), 1, file);
        fclose(file);
    }
    return 0;
}

fseek 和 ftell

Used to move the file pointer in a file and get the position of the file pointer.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        fseek(file, 0, SEEK_END); // 移动到文件末尾
        long size = ftell(file);  // 获取文件大小
        fclose(file);
    }
    return 0;
}

fflush

Flush the file stream to write the data in the buffer to the file.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "w");
    if (file) {
        fputs("Hello, World!", file);
        fflush(file); // 确保数据立即写入文件
        fclose(file);
    }
    return 0;
}

printf 和 fprintf

Used for formatted output to standard output or a file.

实例

#include <cstdio>

int main() {
    int value = 42;
    printf("Value: %d\n", value); // 输出到标准输出

    FILE* file = fopen("output.txt", "w");
    if (file) {
        fprintf(file, "Value: %d\n", value); // 输出到文件
        fclose(file);
    }
    return 0;
}

scanf 和 fscanf

Used for formatted input from standard input or a file.

实例

#include <cstdio>

int main() {
    int value;
    scanf("%d", &value); // 从标准输入读取

    FILE* file = fopen("input.txt", "r");
    if (file) {
        fscanf(file, "%d", &value); // 从文件读取
        fclose(file);
    }
    return 0;
}

sprintf 和 sscanf

Used for formatted output to a string and reading from a string.

实例

#include <cstdio>

int main() {
    char buffer[50];
    int value = 42;
    sprintf(buffer, "Value: %d", value); // 输出到字符串

    int readValue;
    sscanf(buffer, "Value: %d", &readValue); // 从字符串读取

    return 0;
}

fgets 和 fputs

Used to read a string from a file and write a string to a file.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        char buffer[100];
        if (fgets(buffer, sizeof(buffer), file)) {
            // 读取成功
        }
        fclose(file);
    }

    file = fopen("example.txt", "w");
    if (file) {
        fputs("Hello, World!\n", file); // 写入字符串到文件
        fclose(file);
    }
    return 0;
}

getc 和 putc

Used to read characters from a file and write characters to a file.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        int c = getc(file); // 读取一个字符
        fclose(file);
    }

    file = fopen("example.txt", "w");
    if (file) {
        putc('A', file); // 写入一个字符
        fclose(file);
    }
    return 0;
}

feof 和 ferror

Used to detect end-of-file and file errors.

实例


#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        while (!feof(file)) {
            int c = getc(file);
            if (ferror(file)) {
                // 处理错误
                break;
            }
            // 处理字符
        }
        fclose(file);
    }
    return 0;
}

EOF

Indicates the end-of-file flag.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file) {
        int c;
        while ((c = getc(file)) != EOF) {
            // 处理字符
        }
        fclose(file);
    }
    return 0;
}

NULL

Represents a null pointer.

实例

#include <cstdio>

int main() {
    FILE* file = fopen("example.txt", "r");
    if (file == NULL) {
        // 处理文件打开错误
    } else {
        fclose(file);
    }
    return 0;
}

注意事项

  • 使用<cstdio>When using the library, you need to ensure that file open and close are handled correctly to avoid resource leaks.
  • 在使用fopenWhen using it, you need to provide the correct file path and mode.
  • 使用fprintf 和 fscanfWhen using it, the format string should match the variable types.
